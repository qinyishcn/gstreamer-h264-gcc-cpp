cmake_minimum_required(VERSION 3.16)
project(h265_resilience LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.26
    gstreamer-app-1.0 gstreamer-video-1.0 gstreamer-rtp-1.0)
add_library(resilience_core src/resilience.cpp)
target_include_directories(resilience_core PUBLIC include)
target_compile_options(resilience_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core PRIVATE resilience_core)
enable_testing()
add_test(NAME core COMMAND test_core)
add_library(gst_experiment src/gst_experiment.cpp)
target_link_libraries(gst_experiment PUBLIC resilience_core PkgConfig::GST Threads::Threads)
target_compile_options(gst_experiment PRIVATE -Wall -Wextra -Wpedantic)
add_executable(test_pipeline tests/test_pipeline.cpp)
target_link_libraries(test_pipeline PRIVATE gst_experiment)
add_test(NAME pipeline COMMAND test_pipeline)
set_tests_properties(pipeline PROPERTIES TIMEOUT 150)
add_executable(h265_resilience src/main.cpp)
target_link_libraries(h265_resilience PRIVATE gst_experiment)
target_compile_options(h265_resilience PRIVATE -Wall -Wextra -Wpedantic)
add_executable(h265_live src/live.cpp)
target_link_libraries(h265_live PRIVATE gst_experiment)
target_compile_options(h265_live PRIVATE -Wall -Wextra -Wpedantic)
add_executable(h265_report src/report.cpp)
target_compile_options(h265_report PRIVATE -Wall -Wextra -Wpedantic)
if(UNIX)
    add_test(NAME live_udp COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_live.sh
        ${CMAKE_CURRENT_BINARY_DIR} ${CMAKE_CURRENT_BINARY_DIR}/live-test)
    set_tests_properties(live_udp PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
endif()
